cmake_minimum_required(VERSION 3.16)
project(three_body_problem LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

option(ORBIT_BUILD_VIEWER "Build the native OpenGL viewer" ON)
if(ORBIT_BUILD_VIEWER)
    find_package(OpenGL REQUIRED)
    find_package(glfw3 3.3 REQUIRED)
    find_package(GLEW REQUIRED)
    add_executable(three_body_problem src/main.cpp src/hud.cpp src/alien_viewer.cpp)
    target_link_libraries(three_body_problem PRIVATE glfw GLEW::GLEW OpenGL::GL)
    target_compile_definitions(three_body_problem PRIVATE SHADER_DIR="${CMAKE_CURRENT_SOURCE_DIR}/shaders")
endif()
add_executable(orbit_export src/export.cpp)
add_executable(orbit02_export src/alien_export.cpp)
add_executable(tidal_experiment src/tidal_experiment.cpp)
add_executable(stellar_stability src/stability_experiment.cpp)
add_executable(stellar_dance src/stellar_dance.cpp)
add_executable(orbit_catalog_export src/catalog_export.cpp)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    if(ORBIT_BUILD_VIEWER)
        target_compile_options(three_body_problem PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    target_compile_options(orbit_export PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(orbit02_export PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(tidal_experiment PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(stellar_stability PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(stellar_dance PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(orbit_catalog_export PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(catalog_tests tests/catalog_tests.cpp)
    target_include_directories(catalog_tests PRIVATE src)
    add_test(NAME catalog_tests COMMAND catalog_tests)
    add_executable(stability_tests tests/stability_tests.cpp)
    target_include_directories(stability_tests PRIVATE src)
    add_test(NAME stability_tests COMMAND stability_tests)
    add_executable(alien_tests tests/alien_tests.cpp)
    target_include_directories(alien_tests PRIVATE src)
    add_test(NAME alien_tests COMMAND alien_tests)
    add_executable(scene_tests tests/scene_tests.cpp)
    target_include_directories(scene_tests PRIVATE src)
    add_test(NAME scene_tests COMMAND scene_tests)
    add_executable(physics_tests tests/physics_tests.cpp)
    target_include_directories(physics_tests PRIVATE src)
    add_test(NAME physics_tests COMMAND physics_tests)
    add_executable(diagnostics_tests tests/diagnostics_tests.cpp)
    target_include_directories(diagnostics_tests PRIVATE src)
    add_test(NAME diagnostics_tests COMMAND diagnostics_tests)
    add_executable(resonance_tests tests/resonance_tests.cpp)
    target_include_directories(resonance_tests PRIVATE src)
    add_test(NAME resonance_tests COMMAND resonance_tests)
    add_executable(tides_tests tests/tides_tests.cpp)
    target_include_directories(tides_tests PRIVATE src)
    add_test(NAME tides_tests COMMAND tides_tests)
    find_package(Python3 COMPONENTS Interpreter QUIET)
    if(Python3_Interpreter_FOUND)
        add_test(NAME catalog_export_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/catalog_export_tests.py $<TARGET_FILE:orbit_catalog_export>)
        add_test(NAME dance_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/dance_tests.py $<TARGET_FILE:stellar_dance>)
        add_test(NAME alien_export_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/alien_export_tests.py $<TARGET_FILE:orbit02_export>)
        add_test(NAME export_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/export_tests.py $<TARGET_FILE:orbit_export>)
        add_test(NAME tidal_report_tests COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tidal_report_tests.py)
    endif()
    find_program(NODE_EXECUTABLE node)
    if(NODE_EXECUTABLE)
        add_test(NAME alien_replay_tests COMMAND ${NODE_EXECUTABLE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/alien-replay.test.mjs)
        add_test(NAME replay_tests COMMAND ${NODE_EXECUTABLE} --test ${CMAKE_CURRENT_SOURCE_DIR}/tests/replay.test.mjs)
    endif()
endif()
